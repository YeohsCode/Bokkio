def replace_checkpoint(source, target):
    """Windows readers may briefly deny atomic replacement of a checkpoint."""
    for attempt in range(6):
        try:
            os.replace(source, target)
            return
        except OSError as error:
            if getattr(error, "winerror", None) not in {5, 32, 33} or attempt == 5:
                raise
            time.sleep(0.05 * 2 ** attempt)


def verify_conditions(snapshot, checks):
    results = []
    nodes = flatten(snapshot["windows"])
    by_ref = {n["ref"]: n for n in nodes}
    for check in checks:
        candidates = [n for n in nodes if n["role"] == check["role"]
                      and (check["name"] is None or n["name"] == check["name"])
                      and (check.get("parent_name") is None or by_ref.get(n["parent"], {}).get("name") == check["parent_name"])]
        field = check["field"]
        if field == "present":
            passed = (len(candidates) == 1 if check["equals"] else len(candidates) == 0)
            results.append({"condition":check, "candidates":len(candidates), "matched":len(candidates), "passed":passed})
            continue
        unique_candidates = []
        seen = set()
        for candidate in candidates:
            data = candidate['platform_data']
            identity = data.get("value_runtime_id") or data.get('action_runtime_id')
            actual = candidate[field] if field in {"name", "value"} else candidate["state"].get(field)
            key = ((data.get('action_process_id') if data.get('action_runtime_id') else data.get('value_scope_hwnd'), tuple(identity), actual)
                   if identity else (candidate["ref"],))
            if key not in seen:
                unique_candidates.append(candidate); seen.add(key)
        candidates = unique_candidates
        def matches(node):
            actual = node[field] if field in {"name", "value"} else node["state"].get(field)
            wanted = check['equals']
            if field=='checked' and actual in {'on','off'}:
                actual = actual=='on'
            if field=='value' and node['role']=='text_area' and isinstance(actual,str) and isinstance(wanted,str):
                actual, wanted = actual.replace('\r\n','\n'), wanted.replace('\r\n','\n')
            return actual == wanted
        matched = [n for n in candidates if matches(n)]
        # A changing label can identify itself by its expected full name.
        unique = len(matched) == 1 if field == "name" else len(candidates) == 1 and len(matched) == 1
        results.append({"condition": check, "candidates": len(candidates), "matched": len(matched), "passed": unique})
    return bool(results) and all(r["passed"] for r in results), results


def planner_observation(goal, snapshot, windows, limit=240):
    """Compact native context: goal matches first, balanced window samples otherwise."""
    nodes = flatten(snapshot["windows"])
    by_ref = {n["ref"]: n for n in nodes}
    scoped = bounded_observation(goal, snapshot)
    strategy = "full"
    if len(nodes)<=limit:
        selected=nodes
    elif "observation_scope" in scoped:
        selected = flatten(scoped["windows"])
        strategy = "exact_observed_names"
        # Status labels can describe the outcome even when their current names
        # do not occur in the goal. Keep a bounded set beside named controls.
        seen = {n["ref"] for n in selected}
        # Planning needs nearby controls to discover the next step, for example
        # Submit beside Search even when the goal only names Search.
        for node in list(selected):
            parent = by_ref.get(node["parent"])
            if parent is None or len(parent["children"]) > 32: continue
            for sibling in parent["children"]:
                nearby = flatten([sibling])
                if len(nearby) > 32: continue
                for candidate in nearby:
                    if candidate["ref"] not in seen and len(selected) < limit:
                        selected.append(candidate); seen.add(candidate["ref"])
        labels = [n for n in nodes if n["role"] == "static_text" and n["ref"] not in seen]
        selected.extend(labels[:min(32, max(0, limit - len(selected)))])
    elif len(nodes) <= limit:
        selected = nodes
    else:
        strategy = "balanced_windows"
        groups = {}
        for node in nodes:
            parent = node
            while parent["role"] not in {"window", "dialog"} and parent["parent"] in by_ref:
                parent = by_ref[parent["parent"]]
            key = parent["ref"] if parent["role"] in {"window", "dialog"} else None
            groups.setdefault(key, []).append(node)
        # Allocate fairly across windows. Sample each group across its full
        # traversal so a large first window cannot hide all later controls.
        quotas = {key: 0 for key in groups}
        remaining = limit
        while remaining:
            changed = False
            for key, group in groups.items():
                if remaining and quotas[key] < len(group):
                    quotas[key] += 1; remaining -= 1; changed = True
            if not changed: break
        selected = []
        for key, group in groups.items():
            count = quotas[key]
            indices = ([0] if count == 1 else
                       [i * (len(group) - 1) // (count - 1) for i in range(count)])
            selected.extend(group[i] for i in indices)
    rows = []
    for node in selected:
        actions = node['actions']
        data = node['platform_data']
        if data.get('ambiguous_ref_identity'):
            actions = [a for a in actions if a in data.get('verified_native_actions', [])
                       or (data.get('mac_unique_identity') and a in data.get('mac_verified_actions',[]))
                       or (a in {'set_value','submit'} and data.get('value_runtime_id'))]
        if node['state'].get('enabled') is False:
            actions=[a for a in actions if a=='select_text' and data.get('mac_unique_identity')]
        parent = node
        while parent["role"] not in {"window", "dialog"} and parent["parent"] in by_ref:
            parent = by_ref[parent["parent"]]
        rows.append({**({"native_inline_editor":True} if data.get("class_name")=="UIRenameTextElement" else {}),
                     **({'source':'OCR','ocr_confidence':data['ocr_confidence']} if data.get('source')=='OCR' else {}),
                     **({'value_submission':data['value_submission']} if data.get('value_submission') else {}),
                     "role": node["role"], "name": node["name"], "value": node["value"], "actions": actions,
                     "parent_name": by_ref.get(node["parent"], {}).get("name"),
                     "window_name": parent["name"] if parent["role"] in {"window", "dialog"} else None,
                     "state": {k: v for k, v in node["state"].items() if v is not None and k in
                               {"enabled", "visible", "focused", "selected", "selected_text", "expanded", "checked", "editable"}}})
    words = list(dict.fromkeys(re.findall(r'(?:\bword\s*|单词\s*|词\s*)["“\']([^"”\'\n]{1,64})["”\']', goal, re.I)))[:16]
    text_facts = []
    for row in rows:
        if row['role'] != 'text_area' or not isinstance(row['value'], str) or not row['value'] or not words:
            continue
        counts = {}
        for word in words:
            pattern = r'(?<!\w)' + re.escape(word) + r'(?!\w)'
            counts[word] = {'exact_substring':row['value'].count(word),
                            'whole_word_case_sensitive':len(re.findall(pattern, row['value'])),
                            'whole_word_case_insensitive':len(re.findall(pattern, row['value'], re.I))}
        text_facts.append({'name':row['name'], 'parent_name':row['parent_name'], 'window_name':row['window_name'],
                           'basis':'current native text_area value',
                           'native_text_sha256':hashlib.sha256(row['value'].encode('utf-8')).hexdigest(),
                           'characters':len(row['value']), 'word_counts':counts})
    return {"app": snapshot["app"], "windows": [{"name": w["name"]} for w in windows], "nodes": rows,
            "text_facts":text_facts,
            "truncated_nodes": len(nodes) - len(selected),
            "observation_scope": {"strategy": strategy, "original_nodes": len(nodes), "retained_nodes": len(selected), "limit": limit}}


class DesktopAgent:
    def __init__(self, backend, planner, provider, *, max_actions=20, max_replans=1, max_phases=4, control=None, final_verifier=None, required_sources=()):
        if type(max_actions) is not int or type(max_replans) is not int or not 1 <= max_actions <= 100 or not 0 <= max_replans <= 3:
            raise BokkioError("Action budget must be 1–100 and replan budget 0–3")
        if type(max_phases) is not int or not 0 <= max_phases <= 8:
            raise BokkioError("Phase continuation budget must be 0–8")
        self.backend, self.planner, self.provider = backend, planner, provider
        self.max_actions, self.max_replans, self.control = max_actions, max_replans, control
        self.max_phases = max_phases
        self.final_verifier = final_verifier
        if (not isinstance(required_sources,(list,tuple)) or len(required_sources)>16
                or any(not isinstance(p,str) or not p or any(c in p for c in '\0\r\n')
                       or not ((ntpath.splitdrive(p)[0] and ntpath.isabs(p)) or Path(p).is_absolute()) for p in required_sources)):
            raise BokkioError('Required sources need up to 16 absolute file paths')
        self.required_sources=sorted(set(required_sources))

    def _control(self):
        action = self.control() if self.control else None
        if action is not None and (not isinstance(action, str) or action not in {"pause", "cancel"}):
            raise BokkioError("Control must be pause, cancel or null")
        return action

    def _observations(self, apps, goal):
        observations = {}
        for app in apps:
            snapshot = (self.backend.observe_for_planner(app,goal) if hasattr(self.backend,'observe_for_planner')
                        else self.backend.snapshot(app))
            # Window names and nodes must describe the same native observation.
            # windows() previously traversed the complete UIA tree a second time.
            windows = []
            def visit(node):
                if node['role'] in {'window','dialog'}:
                    windows.append({'name':node['name'], 'ref':node['ref']})
                else:
                    for child in node['children']:visit(child)
            for root in snapshot['windows']:visit(root)
            observations[app] = planner_observation(goal, snapshot, windows)
        return observations

    def run(self, goal, apps, output, *, resume=None, plan_only=False, amend_reason=None):
        if not isinstance(goal, str) or not goal.strip() or not apps or any(not isinstance(app, str) or not app.strip() for app in apps) or len(set(apps)) != len(apps):
            raise BokkioError("Run needs a nonempty goal and a unique explicit app allowlist")
        output = Path(output)
        try:
            trace = (json.loads(Path(resume).read_text(encoding="utf-8")) if resume else
                     {"goal": goal, "apps": list(apps), "events": [], "completed_steps": [], "actions": 0, "replans": 0})
        except (OSError, ValueError):
            raise BokkioError("Could not read valid task checkpoint") from None
        if not isinstance(trace, dict) or trace.get("apps") != list(apps):
            raise BokkioError("Resume goal and app allowlist must match the checkpoint")
        changed = trace.get("goal") != goal
        if amend_reason is not None:
            if (not resume or not isinstance(amend_reason, str) or not amend_reason.strip()
                    or not changed or trace.get("status") not in {"paused", "completed"}):
                raise BokkioError("Requirement amendment needs a paused/completed checkpoint, a new goal and a nonempty reason")
        elif changed:
            raise BokkioError("Resume goal and app allowlist must match the checkpoint")
        previous_sources=trace.get('required_sources', self.required_sources if not resume else [])
        if not isinstance(previous_sources,list) or any(not isinstance(p,str) for p in previous_sources):
            raise BokkioError('Invalid source constraint checkpoint')
        if resume and previous_sources != self.required_sources and amend_reason is None:
            raise BokkioError('Resume source constraints must match the checkpoint')
        execution_contract=getattr(self.backend,'execution_contract',None)
        # JSON normalization keeps tuple-based runtime policies comparable to
        # their saved representation and prevents silently widening a resume.
        execution_contract=json.loads(json.dumps(execution_contract))
        if resume and trace.get('execution_contract')!=execution_contract:
            raise BokkioError('Resume visual scopes and policies must match the checkpoint')
        if execution_contract is not None:trace['execution_contract']=execution_contract
        if hasattr(self.backend,'begin_run'):
            if resume and not isinstance(trace.get('visual_capture_counts'),dict):
                raise BokkioError('Visual resume requires its observation budget checkpoint')
            self.backend.begin_run(trace.get('visual_capture_counts') if resume else None)
        if (not isinstance(trace.get("events"), list) or not isinstance(trace.get("completed_steps"), list)
                or type(trace.get("actions")) is not int or trace["actions"] < 0
                or type(trace.get("replans")) is not int or trace["replans"] < 0):
            raise BokkioError("Invalid task checkpoint")
        trace.setdefault("phases", 0)
        if type(trace["phases"]) is not int or trace["phases"] < 0:
            raise BokkioError("Invalid phase checkpoint")
        revision = trace.setdefault("goal_revision", 1)
        requirements = trace.setdefault("requirements", [{"revision":revision, "goal":trace["goal"], 'required_sources':previous_sources}])
        if (type(revision) is not int or revision < 1 or not isinstance(requirements, list)
                or len(requirements) != revision
                or any(not isinstance(r, dict) or r.get("revision") != i+1
                       or not isinstance(r.get("goal"), str) or not r["goal"].strip()
                       for i, r in enumerate(requirements))
                or requirements[-1]["goal"] != trace["goal"]
                or not isinstance(trace.setdefault("artifact_versions", []), list)):
            raise BokkioError("Invalid requirement revision checkpoint")
        def event(kind, **data):
            if hasattr(self.backend,'visual_capture_counts'):
                trace['visual_capture_counts']=self.backend.visual_capture_counts
            trace["events"].append({"kind": kind, "time": time.time(), "goal_revision":trace["goal_revision"], **data})
            temporary = None
            try:
                output.parent.mkdir(parents=True, exist_ok=True)
                with tempfile.NamedTemporaryFile(mode="w", encoding="utf-8", dir=output.parent, delete=False) as file:
                    temporary = file.name
                    json.dump(trace, file, ensure_ascii=False, indent=2)
                    file.flush()
                    os.fsync(file.fileno())
                replace_checkpoint(temporary, output)
            except OSError:
                raise BokkioError("Could not write private task trace") from None
            finally:
                if temporary and os.path.exists(temporary): os.unlink(temporary)
        def stop(status, reason=None):
            trace["status"] = status
            try:
                event("status", status=status, reason=reason)
            except BokkioError as error:
                # A full disk or persistent lock can also prevent recording the
                # failure. Return it in memory without dispatching more actions.
                trace["status"] = "failed"
                trace["checkpoint_error"] = str(error)
                trace["events"][-1].update(status="failed", reason=str(error))
            return trace
        def completed(step):
            # IDs alone are insufficient: a replan can reuse an ID for different
            # work. Retain the whole verified subtask as the execution receipt.
            return any(e.get("kind") == "subtask_completed" and e.get("subtask") == step
                       and e.get("goal_revision", 1) == trace["goal_revision"]
                       for e in trace["events"])

        def dispatched(step):
            return any(e.get('kind')=='action' and e.get('subtask')==step
                       and e.get('goal_revision',1)==trace['goal_revision'] for e in trace['events'])

        def receipts():
            return [e["subtask"] for e in trace["events"] if e.get("kind") == "subtask_completed"
                    and "subtask" in e and e.get("goal_revision", 1) == trace["goal_revision"]]

        def missing_sources():
            acquired={e.get('path') for e in trace['events'] if e.get('kind')=='source_acquired'
                      and e.get('goal_revision',1)==trace['goal_revision']}
            return [p for p in self.required_sources if p not in acquired]

        def source_for_open(target, snapshot):
            if not self.required_sources or target['role']!='button' or target['name']!='Open': return None
            nodes={n['ref']:n for n in flatten(snapshot['windows'])}
            parent=target
            while parent['role'] not in {'dialog','window'} and parent['parent'] in nodes:
                parent=nodes[parent['parent']]
            if (target['platform']!='windows' or parent['role']!='dialog' or parent['name']!='Open'
                    or parent['platform_data'].get('class_name')!='#32770'):
                raise BokkioError('Required source acquisition supports Windows native Open dialogs')
            fields=[n for n in flatten(parent['children']) if n['role']=='text_field'
                    and (n['name'] or '').strip().rstrip(':').casefold()=='file name'
                    and isinstance(n['value'],str) and n['platform_data'].get('value_source')=='UIA.ValuePattern']
            identities={(n['platform_data'].get('value_scope_hwnd'),tuple(n['platform_data'].get('value_runtime_id') or [n['ref']])) for n in fields}
            values={ntpath.normcase(ntpath.normpath(n['value'].strip('"'))) for n in fields}
            matches=[p for p in self.required_sources if ntpath.normcase(ntpath.normpath(p)) in values]
            if len(matches)!=1 or len(values)!=1 or len(identities)!=1:
                raise BokkioError('Before native Open, set File name: to one full required source path: '+json.dumps(self.required_sources))
            return matches[0]

        def complete(step):
            if step["id"] not in trace["completed_steps"]:
                trace["completed_steps"].append(step["id"])
            event("subtask_completed", step=step["id"], subtask=step)
        def make_plan(observations, progress):
            inline=[{'app':app,**node} for app,obs in observations.items() for node in obs['nodes']
                    if node.get('native_inline_editor') and node['state'].get('editable')]
            if inline:
                progress={**(progress or {}),'native_inline_editor_opened':'Rename','new_editors':inline}
            if self.required_sources:
                source_facts={e['path']:{k:e[k] for k in ['path','app','native_text_sha256','text_facts'] if k in e}
                              for e in trace['events'] if e.get('kind')=='source_acquired'
                              and e.get('goal_revision',1)==trace['goal_revision']}
                progress={**(progress or {}), 'required_sources':self.required_sources,
                          'missing_native_sources':missing_sources(), 'native_sources':list(source_facts.values())}
            while True:
                plan, metadata = self.planner.plan(goal, observations, progress)
                event("plan_proposal", plan=plan, model=metadata)
                try:
                    # Selection literals are read-only: bind an omitted value
                    # only when the predicate names one actual observed string.
                    # Preserve the raw proposal as an immutable event.
                    plan=copy.deepcopy(plan)
                    if isinstance(plan,dict) and isinstance(plan.get('steps'),list):
                        for proposed in plan['steps']:
                            if not isinstance(proposed,dict) or not re.search(r'\bselect\b',str(proposed.get('goal','')),re.I):continue
                            values=proposed.get('allowed_values')
                            if not isinstance(values,list):continue
                            for check in proposed.get('success',[]):
                                if not isinstance(check,dict) or check.get('field')!='selected_text':continue
                                literal=check.get('equals')
                                rows=observations.get(proposed.get('app'),{}).get('nodes',[])
                                matches=[n for n in rows if n['role']==check.get('role') and n['name']==check.get('name')
                                         and isinstance(n['value'],str) and isinstance(literal,str) and literal and n['value'].count(literal)==1]
                                if len(matches)==1 and literal not in values and len(values)<16:
                                    values.append(literal)
                                    event('selection_literal_bound',step=proposed.get('id'),source='unique_current_native_text')
                    validated=validate_plan(plan, apps, observations)
                    if progress and progress.get('native_inline_editor_opened') and validated['steps']:
                        first=validated['steps'][0]
                        if re.search(r"\b(?:click|press|invoke)\s+(?:the\s+)?['\"]?rename\b",first['goal'],re.I):
                            raise BokkioError('Inline rename editor is already open. Use the observed new editor to write the filename, then submit and verify the renamed row. Do not invoke Rename again.')
                    if progress and progress.get('native_combo_value_written'):
                        first=validated['steps'][0];target=progress['native_combo_value_written']
                        if (not first.get('requires_action') or not re.search(r'\b(?:submit|confirm|commit)\b',first['goal'],re.I)
                                or target['name'].casefold().rstrip(':') not in first['goal'].casefold()):
                            raise BokkioError('The Mac combo value was written but not committed. Next subtask must explicitly confirm '+target['name']+' using submit, requires_action=true, allowed_values=[], then observe the result.')
                    return validated, metadata
                except BokkioError as error:
                    event("error", error=str(error), stage="plan_validation")
                    if trace["replans"] >= self.max_replans:
                        return None, str(error)
                    trace["replans"] += 1
                    progress = {**(progress or {}), "error":str(error), "invalid_plan":plan}
        amendment = None
        if changed:
            amendment = {"previous_goal":trace["goal"], "new_goal":goal, "reason":amend_reason,
                         "previous_revision":revision}
            trace["goal_revision"] += 1
            trace["goal"] = goal
            requirements.append({"revision":trace["goal_revision"], "goal":goal, "reason":amend_reason,
                                 'required_sources':self.required_sources})
            # Old receipts remain in history, scoped to the old requirement.
            # Cumulative action, recovery and phase budgets remain in force.
            trace.pop("plan", None)
            trace["completed_steps"] = []
        trace['required_sources']=self.required_sources
        trace["status"] = "running"
        if amendment:
            event("requirement_amended", **amendment)
        event("start", resumed=bool(resume))
        try:
            control = self._control()
            if control: return stop("paused" if control == "pause" else "cancelled")
            if "plan" not in trace:
                observations = self._observations(apps, goal)
                event("planner_observations", observations=observations)
                progress = {"requirement_change":amendment} if amendment else None
                plan, metadata = make_plan(observations, progress)
                if plan is None: return stop("blocked", metadata)
                trace["plan"] = plan
                event("plan", plan=plan, model=metadata)
            else:
                validate_plan(trace["plan"], apps)
            if plan_only: return stop("planned")
            while True:
                restarted = False
                for index, step in enumerate(trace["plan"]["steps"]):
                    # Intermediate success may intentionally be overwritten by
                    # later steps. Resume from its receipt, rather than repeat
                    # the write. The final step still needs a fresh observation.
                    if index < len(trace["plan"]["steps"]) - 1 and completed(step):
                        event("subtask_retained", step=step["id"], evidence="verified_history")
                        continue
                    seen_states = {}
                    stale_refreshes=0
                    while True:
                        control = self._control()
                        if control: return stop("paused" if control == "pause" else "cancelled")
                        if step["risk"] in {"external", "destructive"}:
                            return stop("blocked", "Sensitive subtask requires explicit approval outside this loop")
                        calls = []
                        try:
                            snapshot = self.backend.snapshot(step["app"], step["window"])
                            if hasattr(self.backend,'prepare_step'):
                                snapshot=self.backend.prepare_step(step['app'],snapshot,step)
                            passed, checks = verify_conditions(snapshot, step["success"])
                            event("observation", step=step["id"], snapshot=snapshot, checks=checks)
                            if passed and (not step.get('requires_action',False) or dispatched(step)):
                                complete(step)
                                break
                            for check_result in checks:
                                condition=check_result['condition']
                                if condition['field']=='checked' and check_result['candidates']>1:
                                    by_ref={n['ref']:n for n in flatten(snapshot['windows'])}
                                    observed=[{'parent_name':by_ref.get(n['parent'],{}).get('name'),
                                               'checked':n['state'].get('checked')}
                                              for n in by_ref.values() if n['role']==condition['role']
                                              and (condition['name'] is None or n['name']==condition['name'])]
                                    raise BokkioError('Checked success condition is ambiguous; replan using the observed immediate parent_name. Candidates: '+json.dumps(observed,ensure_ascii=False))
                            digest=tree_digest(snapshot['windows'])
                            previous_actions=seen_states.get(digest)
                            if previous_actions is not None and trace['actions']-previous_actions >= 2:
                                raise BokkioError('Native UI returned to an earlier state without satisfying this subtask. Repair conflicting success conditions or the action route; do not repeat this cycle.')
                            seen_states.setdefault(digest,trace['actions'])
                            # A requested state already holds under the real parent.
                            # Do not toggle it away to chase an invented menu hierarchy.
                            for check in step['success']:
                                if check['field']=='checked' and check.get('parent_name') is not None:
                                    relaxed={**check, 'parent_name':None}
                                    already,_=verify_conditions(snapshot,[relaxed])
                                    exact,_=verify_conditions(snapshot,[check])
                                    if already and not exact:
                                        raise BokkioError('Native checked state already matches; parent_name does not match the observed immediate parent. Replan with the actual parent or null; do not toggle the setting again.')
                            if trace["actions"] >= self.max_actions: return stop("blocked", "Action budget exhausted")
                            provider = self.provider
                            class Capture:
                                def ask(self, state, questions):
                                    response = provider.ask(state, questions)
                                    calls.append(response)
                                    return response
                            decision_goal = step["goal"] + "\nRequired observable outcome: " + json.dumps(step["success"], ensure_ascii=False)
                            decision, response = decide(Capture(), decision_goal, snapshot, step["allowed_values"], allow_done=False,
                                                        require_action=step.get('requires_action',False) and not dispatched(step))
                            event("decision", step=step["id"], decision=decision.as_dict(), model_calls=calls)
                            control = self._control()
                            if control: return stop("paused" if control == "pause" else "cancelled")
                            if decision.action == "done": raise BokkioError("Model completion lacks independent success evidence")
                            target = next(n for n in flatten(snapshot["windows"]) if n["ref"] == decision.ref)
                            source_path=source_for_open(target,snapshot) if decision.action=='click' else None
                            if decision.action in {'set_value','type'} and target['role']=='text_area' and missing_sources():
                                raise BokkioError('Read the required sources through native Open before writing output: '+json.dumps(missing_sources()))
                            if decision.action in {"click", "select", "set_value", "type", "visual_click", "visual_type"} and re.search(r"\b(send|delete|pay|purchase|transfer|publish)\b|发送|删除|支付|转账|发布", target.get("name") or "", re.I):
                                return stop("blocked", "Sensitive native target requires explicit approval")
                            trace["actions"] += 1
                            result = execute_decision(self.backend, step["app"], decision)
                            event("action", step=step["id"], subtask=step, result=result)
                            popup_closed=False
                            try:fresh = self.backend.snapshot(step["app"], step["window"])
                            except BokkioError as error:
                                if step['window'] is None or not str(error).startswith('Window not found:'):raise
                                fresh=self.backend.snapshot(step['app'])
                                if any(n['role'] in {'window','dialog'} and n['name']==step['window'] for n in flatten(fresh['windows'])):raise
                                popup_closed=True
                                event('window_scope_changed',previous_window=step['window'],snapshot=fresh)
                            if source_path:
                                current=flatten(fresh['windows'])
                                editors=[n for n in current if n['role']=='text_area' and isinstance(n['value'],str)]
                                stem=ntpath.splitext(ntpath.basename(source_path))[0].casefold()
                                source_title=any(n['role']=='window' and stem in (n['name'] or '').casefold() for n in current)
                                if any(n['role']=='dialog' for n in current) or len(editors)!=1 or not source_title:
                                    raise BokkioError('Required source Open has no unique native editor readback')
                                event('source_acquired',path=source_path,app=step['app'],editor=editors[0]['ref'],
                                      native_text_sha256=hashlib.sha256(editors[0]['value'].encode('utf-8')).hexdigest(),
                                      text_facts=planner_observation(goal,fresh,[])['text_facts'])
                            passed, checks = verify_conditions(fresh, step["success"])
                            event("outcome", step=step["id"], snapshot=fresh, passed=passed, checks=checks)
                            combo_written=(decision.action=='set_value' and target['platform']=='macos'
                                           and target['role']=='combo_box' and 'submit' in target['actions'] and not result.get('input_commit'))
                            visual_committed=bool(result.get('input_commit')) and not passed
                            if passed and not combo_written:
                                complete(step)
                            menu_changed=(decision.action=='click' and target['role']=='menu_item'
                                          and 'toggle' in [a.casefold() for a in target['platform_data'].get('actions',[])]
                                          and target['state'].get('checked') is not None and result['tree_changed'])
                            def editors(state):
                                return {tuple(n['platform_data'].get('value_runtime_id') or [n['ref']])
                                        for n in flatten(state['windows']) if n['role']=='text_field'
                                        and n['state'].get('editable') and n['state'].get('visible') is not False}
                            rename_editor_opened=(decision.action=='click' and target['role']=='button'
                                                  and (target['name'] or '').casefold()=='rename'
                                                  and bool(editors(fresh)-editors(snapshot)))
                            address_submitted=(decision.action=='submit' and (target['name'] or '').casefold()=='address bar'
                                               and not passed and
                                               {w.get('name') for w in flatten(fresh['windows']) if w['role']=='window'}!=
                                               {w.get('name') for w in flatten(snapshot['windows']) if w['role']=='window'})
                            popup_opened=(decision.action=='expand' and target['platform']=='macos'
                                          and result.get('action_source')=='AXShowMenu'
                                          and (result.get('verification')=='confirmed' or
                                               {n['name'] for n in flatten(fresh['windows']) if n['role'] in {'window','dialog'}} !=
                                               {n['name'] for n in flatten(snapshot['windows']) if n['role'] in {'window','dialog'}}))
                            if menu_changed or rename_editor_opened or address_submitted or combo_written or popup_opened or popup_closed or visual_committed:
                                if trace['phases'] >= self.max_phases:
                                    return stop('blocked','Native menu phase budget exhausted')
                                observations=self._observations(apps,goal)
                                event('planner_observations',observations=observations)
                                reason=('visual_value_committed' if visual_committed else 'native_combo_value_written' if combo_written else 'native_popup_closed' if popup_closed else 'native_popup_opened' if popup_opened else
                                        'native_inline_editor_opened' if rename_editor_opened else
                                        'native_address_submitted' if address_submitted else 'native_menu_changed')
                                progress={reason:target['name'], 'completed_steps':trace['completed_steps'],
                                          'completed_subtasks':receipts()}
                                if combo_written:progress['native_combo_value_written']={'name':target['name'],'value':decision.value,'app':step['app']}
                                if visual_committed:
                                    after=result.get('after') or {}
                                    progress['visual_value_committed']={'name':target['name'],'requested':decision.value,'native_value':after.get('value'),
                                                                       'confirmation_posted':True,'property_verification':'final_artifact_required'}
                                if popup_closed:progress['native_popup_closed']={'action':decision.action,'target':target['name'],'intent':step['goal'],
                                                                               'command_dispatched':True,'property_verification':'final_artifact_required'}
                                if address_submitted:
                                    progress['navigation_result']={'submitted_address':target['value'],
                                        'window_names':[n['name'] for n in flatten(fresh['windows']) if n['role']=='window'],
                                        'command_dispatched':True}
                                if rename_editor_opened:
                                    progress['new_editors']=[{'name':n['name'],'value':n['value'],'actions':n['actions'],
                                        'parent_name':None,'state':n['state']} for n in flatten(fresh['windows'])
                                        if n['role']=='text_field' and n['state'].get('editable')
                                        and tuple(n['platform_data'].get('value_runtime_id') or [n['ref']]) in editors(fresh)-editors(snapshot)]
                                plan,metadata=make_plan(observations,progress)
                                if plan is None:return stop('blocked',metadata)
                                trace['plan']=plan;trace['phases']+=1
                                event('phase_plan',plan=plan,model=metadata,phase=trace['phases'],reason=reason)
                                restarted=True
                                break
                            if passed:
                                break
                            # Multi-action dropdown/tree subtasks continue from fresh state.
                        except BokkioError as error:
                            diagnostic=native_error_details(error)
                            event("error", step=step["id"], error=str(error), model_calls=calls, native_error=diagnostic)
                            if isinstance(error,BokkioCompletionUnknown):return stop('blocked','Native completion is unknown; inspect the current UI/artifact before another attempt: '+str(error))
                            if isinstance(error,BokkioStaleSnapshot) and getattr(self.backend,'supports_stale_refresh',False):
                                stale_refreshes+=1
                                event('stale_decision_rejected',step=step['id'],dispatched=False,refresh=stale_refreshes)
                                if stale_refreshes<=2:continue
                                return stop('blocked','Native snapshot changed repeatedly before input')
                            control = self._control()
                            if control: return stop("paused" if control == "pause" else "cancelled")
                            if trace["replans"] >= self.max_replans: return stop("blocked", str(error))
                            trace["replans"] += 1
                            observations = self._observations(apps, goal)
                            event("planner_observations", observations=observations)
                            progress = {"completed_steps": trace["completed_steps"], "error": str(error), "failed_step": step,
                                        "completed_subtasks": receipts(), "native_error": diagnostic}
                            plan, metadata = make_plan(observations, progress)
                            if plan is None: return stop("blocked", metadata)
                            trace["plan"] = plan
                            event("replan", plan=plan, model=metadata)
                            restarted = True
                            break
                    if restarted: break
                if not restarted:
                    control = self._control()
                    if control: return stop("paused" if control == "pause" else "cancelled")
                    if not trace["plan"].get("continue_after_steps", False):
                        missing=missing_sources()
                        if self.final_verifier is None and not missing: return stop("completed")
                        result = ((False,{'missing_native_sources':missing}) if missing else self.final_verifier(trace))
                        if not isinstance(result, (tuple, list)) or len(result) != 2:
                            raise BokkioError("Final verifier must return a boolean and evidence object")
                        passed, details = result
                        if type(passed) is not bool or not isinstance(details, dict):
                            raise BokkioError("Final verifier must return a boolean and evidence object")
                        event("delivery_verification", passed=passed, evidence=details)
                        control = self._control()
                        if control: return stop("paused" if control == "pause" else "cancelled")
                        if passed:
                            version={"goal_revision":trace["goal_revision"], "evidence":details}
                            if not trace['artifact_versions'] or trace['artifact_versions'][-1] != version:
                                trace["artifact_versions"].append(version)
                            return stop("completed")
                        if trace["replans"] >= self.max_replans:
                            return stop("blocked", "Final delivery is not verified")
                        trace["replans"] += 1
                        observations = self._observations(apps, goal)
                        event("planner_observations", observations=observations)
                        progress = {"error":"Final delivery is not verified", "final_verification":details,
                                    "completed_steps":trace["completed_steps"],
                                    "completed_subtasks":receipts()}
                        plan, metadata = make_plan(observations, progress)
                        if plan is None: return stop("blocked", metadata)
                        trace["plan"] = plan
                        event("replan", plan=plan, model=metadata)
                        continue
                    if trace["phases"] >= self.max_phases: return stop("blocked", "Phase continuation budget exhausted")
                    observations = self._observations(apps, goal)
                    event("planner_observations", observations=observations)
                    progress = {"phase_completed": True, "completed_steps": trace["completed_steps"],
                                "completed_subtasks": receipts()}
                    plan, metadata = make_plan(observations, progress)
                    if plan is None: return stop("blocked", metadata)
                    trace["plan"] = plan
                    trace["phases"] += 1
                    event("phase_plan", plan=plan, model=metadata, phase=trace["phases"])
        except KeyboardInterrupt:
            return stop("cancelled", "Interrupted by user")
        except (BokkioError, OSError, ValueError) as error:
            return stop("failed", str(error))


def file_control(path):
    def read():
        if path is None or not Path(path).exists(): return None
        try:
            data = json.loads(Path(path).read_text(encoding="utf-8"))
            if not isinstance(data, dict) or set(data) != {"action"}: raise ValueError()
            return data["action"]
        except (OSError, ValueError): raise BokkioError("Invalid task control file") from None
    return read
