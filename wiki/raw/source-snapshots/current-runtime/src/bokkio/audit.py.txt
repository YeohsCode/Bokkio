"""Persistent execution events. Private requests are never exported as audit data."""
from __future__ import annotations

import csv
import getpass
import hashlib
import io
import json
import os
import platform
from pathlib import Path
import sqlite3
import time
import uuid


def identity():
    """OS identity, not caller arguments or spoofable USER/LOGNAME variables."""
    if os.name != "nt":
        import pwd
        return str(os.getuid()) + ":" + pwd.getpwuid(os.getuid()).pw_name
    import ctypes
    from ctypes import wintypes
    length = wintypes.DWORD(512)
    value = ctypes.create_unicode_buffer(512)
    if not ctypes.windll.secur32.GetUserNameExW(2, value, ctypes.byref(length)):
        raise OSError("Cannot establish Windows caller identity")
    return value.value


def canonical(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False)


def digest(value):
    return hashlib.sha256(canonical(value).encode()).hexdigest()


def identifier():
    return uuid.uuid4().hex


def summarize(value):
    """Log shape, lengths and hashes, never arbitrary input/code/output bodies."""
    if isinstance(value, dict):
        return {str(k): summarize(v) for k, v in value.items()}
    if isinstance(value, list):
        return {"count": len(value), "sha256": digest(value)}
    if isinstance(value, str):
        return {"characters": len(value), "sha256": hashlib.sha256(value.encode()).hexdigest()}
    return value


class AuditStore:
    def __init__(self, root):
        self.root = Path(root).expanduser().resolve()
        self.root.mkdir(parents=True, exist_ok=True, mode=0o700)
        self.path = self.root / "runtime.sqlite3"
        with self.connect() as db:
            db.executescript("""
                CREATE TABLE IF NOT EXISTS sessions (
                    id TEXT PRIMARY KEY, created REAL, expires REAL, owner TEXT,
                    policy TEXT, status TEXT);
                CREATE TABLE IF NOT EXISTS operations (
                    id TEXT PRIMARY KEY, session_id TEXT, tool TEXT, hash TEXT,
                    request TEXT, status TEXT, result TEXT, created REAL);
                CREATE TABLE IF NOT EXISTS approvals (
                    id TEXT PRIMARY KEY, operation_id TEXT UNIQUE, hash TEXT,
                    status TEXT, created REAL, expires REAL, actor TEXT, reason TEXT);
                CREATE TABLE IF NOT EXISTS events (
                    seq INTEGER PRIMARY KEY AUTOINCREMENT, event_id TEXT UNIQUE,
                    time REAL, session_id TEXT, operation_id TEXT, kind TEXT, data TEXT);
                CREATE INDEX IF NOT EXISTS events_operation ON events(operation_id, seq);
                CREATE INDEX IF NOT EXISTS events_session ON events(session_id, seq);
                CREATE TABLE IF NOT EXISTS artifacts (
                    id TEXT PRIMARY KEY, session_id TEXT, operation_id TEXT,
                    path TEXT, sha256 TEXT, mime TEXT, bytes INTEGER);
            """)
        if os.name != "nt":
            os.chmod(self.root, 0o700)
            os.chmod(self.path, 0o600)

    def connect(self):
        db = sqlite3.connect(self.path, timeout=15)
        db.row_factory = sqlite3.Row
        db.execute("PRAGMA busy_timeout=15000")
        db.execute("PRAGMA synchronous=FULL")
        return db

    def emit(self, kind, *, session_id=None, operation_id=None, data=None, db=None):
        if db is not None:
            session = db.execute("SELECT policy FROM sessions WHERE id=?", (session_id,)).fetchone() if session_id else None
            scope = json.loads(session["policy"]) if session else {}
            common = {"platform": platform.system(), "architecture": platform.machine(), "actor": identity(),
                      "device_id": digest({"node": platform.node(), "platform": platform.system()})[:20],
                      "task_id": scope.get("task_id")}
            values = (identifier(), time.time(), session_id, operation_id, kind, canonical({**common, **(data or {})}))
            return db.execute("INSERT INTO events(event_id,time,session_id,operation_id,kind,data) VALUES(?,?,?,?,?,?)", values).lastrowid
        with self.connect() as connection:
            return self.emit(kind, session_id=session_id, operation_id=operation_id, data=data, db=connection)

    def query(self, *, after=0, session_id=None, operation_id=None, kind=None, tool=None, status=None,
              platform=None, error_type=None, task_id=None, since=None, until=None, limit=500):
        if type(after) is not int or after < 0 or type(limit) is not int or not 1 <= limit <= 1000:
            raise ValueError("Invalid event cursor or limit")
        conditions, values = ["seq > ?"], [after]
        for key, value in (("session_id", session_id), ("operation_id", operation_id), ("kind", kind)):
            if value is not None:
                conditions.append(key + " = ?")
                values.append(value)
        for key, value in (("tool", tool), ("status", status), ("platform", platform), ("error_type", error_type)):
            if value is not None:
                conditions.append("json_extract(data, '$." + key + "') = ?")
                values.append(value)
        if task_id is not None:
            conditions.append("session_id IN (SELECT id FROM sessions WHERE json_extract(policy,'$.task_id')=?)")
            values.append(task_id)
        for operator, value in ((">=", since), ("<=", until)):
            if value is not None:
                conditions.append("time " + operator + " ?")
                values.append(float(value))
        with self.connect() as db:
            rows = db.execute("SELECT * FROM events WHERE " + " AND ".join(conditions) + " ORDER BY seq LIMIT ?", (*values, limit)).fetchall()
        return [{**dict(row), "data": json.loads(row["data"])} for row in rows]

    def stats(self, session_id=None):
        where, args = (" WHERE session_id=?", (session_id,)) if session_id else ("", ())
        with self.connect() as db:
            operations = db.execute("SELECT status,tool,COUNT(*) n FROM operations" + where + " GROUP BY status,tool", args).fetchall()
            events = db.execute("SELECT kind,data FROM events" + where, args).fetchall()
        kinds, durations, waits, channels, attempts, dispatched = {}, [], [], {}, 0, 0
        for event in events:
            kinds[event["kind"]] = kinds.get(event["kind"], 0) + 1
            data = json.loads(event["data"])
            if isinstance(data.get("duration_ms"), (int, float)) and (event["kind"] == "job.finished" or data.get("status") in {"executed", "verified", "completed", "failed", "completion_unknown"}):
                durations.append(data["duration_ms"])
            if isinstance(data.get("wait_ms"), (int, float)):
                waits.append(data["wait_ms"])
            if event["kind"] in {"request.received", "request.cached", "request.rejected"}:
                attempts += 1
            if event["kind"] in {"execution.finished", "job.finished"} and data.get("dispatched") is True:
                dispatched += 1
            if event["kind"] == "execution.finished":
                channel = data.get("channel", "unknown")
                channels[channel] = channels.get(channel, 0) + 1
        durations.sort()
        def percentile(p):
            return durations[min(len(durations) - 1, int((len(durations) - 1) * p))] if durations else None
        return {"logical_operations": sum(r["n"] for r in operations),
                "by_tool_status": [dict(r) for r in operations], "events": kinds,
                "attempts": attempts, "known_dispatched": dispatched, "approval_wait_ms": waits,
                "execution_attempts": kinds.get("request.received", 0),
                "cached_responses": kinds.get("request.cached", 0), "rejected_requests": kinds.get("request.rejected", 0),
                "execution_channels": channels, "execution_duration_ms": {"p50": percentile(.5), "p95": percentile(.95)},
                "model_tokens": None, "business_success_rate": None}

    def export(self, format="jsonl", **filters):
        records, cursor = [], filters.pop("after", 0)
        while True:
            page = self.query(after=cursor, limit=1000, **filters)
            records.extend(page)
            if len(page) < 1000:
                break
            cursor = page[-1]["seq"]
        if format == "jsonl":
            return "".join(canonical(r) + "\n" for r in records)
        if format != "csv":
            raise ValueError("Export format must be jsonl or csv")
        buffer = io.StringIO()
        writer = csv.DictWriter(buffer, fieldnames=["seq", "event_id", "time", "session_id", "operation_id", "kind", "data"])
        writer.writeheader()
        writer.writerows({**r, "data": canonical(r["data"])} for r in records)
        return buffer.getvalue()

    def decide(self, approval_id, decision, reason=""):
        """Human/admin entry point; deliberately not registered as an Agent tool."""
        if decision not in {"approved", "denied", "revoked"}:
            raise ValueError("Invalid approval decision")
        with self.connect() as db:
            db.execute("BEGIN IMMEDIATE")
            row = db.execute("SELECT * FROM approvals WHERE id=?", (approval_id,)).fetchone()
            if row is None:
                raise ValueError("Approval not found")
            if row["expires"] <= time.time():
                raise ValueError("Approval expired")
            valid = row["status"] == "pending" if decision != "revoked" else row["status"] == "approved"
            if not valid:
                raise ValueError("Approval is no longer eligible for this decision")
            owner = identity()
            db.execute("UPDATE approvals SET status=?,actor=?,reason=? WHERE id=?", (decision, owner, reason, approval_id))
            op = db.execute("SELECT session_id FROM operations WHERE id=?", (row["operation_id"],)).fetchone()
            self.emit("approval." + decision, session_id=op["session_id"], operation_id=row["operation_id"],
                      data={"approval_id": approval_id, "actor": owner, "reason": reason, "request_sha256": row["hash"]}, db=db)
        return {"approval_id": approval_id, "status": decision, "actor": owner}

    def approvals(self, approval_id=None, session_id=None):
        with self.connect() as db:
            db.execute("BEGIN IMMEDIATE")
            conditions, params = [], []
            if approval_id:
                conditions.append("a.id=?"); params.append(approval_id)
            if session_id:
                conditions.append("o.session_id=?"); params.append(session_id)
            query = "SELECT a.*,o.tool,o.session_id FROM approvals a JOIN operations o ON o.id=a.operation_id" + (
                " WHERE " + " AND ".join(conditions) if conditions else "") + " ORDER BY a.created"
            rows = db.execute(query, params).fetchall()
            for row in rows:
                if row["status"] in {"pending", "approved"} and row["expires"] <= time.time():
                    db.execute("UPDATE approvals SET status='expired' WHERE id=?", (row["id"],))
                    self.emit("approval.expired", session_id=row["session_id"], operation_id=row["operation_id"],
                              data={"approval_id": row["id"], "request_sha256": row["hash"]}, db=db)
            rows = db.execute(query, params).fetchall()
        return [dict(r) for r in rows]
